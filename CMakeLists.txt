cmake_minimum_required(VERSION 3.20)
project(ppwm VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11 xft xres fontconfig)
add_library(ppwm_core STATIC src/ppwm.cpp)
target_link_libraries(ppwm_core PUBLIC PkgConfig::X11)
add_executable(ppwm src/main.cpp)
target_link_libraries(ppwm PRIVATE ppwm_core)
target_compile_options(ppwm_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo)
if(ipo)
 set_property(TARGET ppwm ppwm_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(smart_border_test tests/smart_border_test.cpp)
 target_include_directories(smart_border_test PRIVATE src)
 target_link_libraries(smart_border_test PRIVATE PkgConfig::X11)
 add_executable(slstatus_test tests/slstatus_test.cpp)
 target_link_libraries(slstatus_test PRIVATE PkgConfig::X11)
 add_executable(fullscreen_state_test tests/fullscreen_state_test.cpp)
 target_link_libraries(fullscreen_state_test PRIVATE PkgConfig::X11)
 add_executable(status_test tests/status_test.cpp)
 target_include_directories(status_test PRIVATE src)
 target_link_libraries(status_test PRIVATE PkgConfig::X11)
 add_test(NAME status COMMAND status_test)
 add_executable(config_test tests/config_test.cpp)
 target_include_directories(config_test PRIVATE src)
 target_link_libraries(config_test PRIVATE PkgConfig::X11)
 add_test(NAME config COMMAND config_test)
 add_executable(session_test tests/session_test.cpp)
 target_link_libraries(session_test PRIVATE PkgConfig::X11)
 add_executable(tag_test tests/tag_test.cpp)
 target_link_libraries(tag_test PRIVATE PkgConfig::X11)
 add_executable(xephyr_host tests/xephyr_host.cpp)
 target_link_libraries(xephyr_host PRIVATE PkgConfig::X11)
 pkg_check_modules(XFIXES QUIET IMPORTED_TARGET xfixes)
 if(XFIXES_FOUND)
  add_executable(cursor_test tests/cursor_test.cpp)
  target_link_libraries(cursor_test PRIVATE PkgConfig::X11 PkgConfig::XFIXES)
 endif()
 add_executable(process_test tests/process_test.cpp)
 target_include_directories(process_test PRIVATE src)
 target_link_libraries(process_test PRIVATE ppwm_core)
 add_test(NAME process COMMAND process_test)
 add_executable(workspace_test tests/workspace_test.cpp)
 target_include_directories(workspace_test PRIVATE src)
 target_link_libraries(workspace_test PRIVATE PkgConfig::X11)
 add_test(NAME workspace COMMAND workspace_test)
 add_executable(layout_test tests/layout_test.cpp)
 target_include_directories(layout_test PRIVATE src)
 target_link_libraries(layout_test PRIVATE PkgConfig::X11)
 add_test(NAME layout COMMAND layout_test)
 add_executable(x11_test tests/x11_test.cpp)
 target_include_directories(x11_test PRIVATE src)
 target_link_libraries(x11_test PRIVATE PkgConfig::X11)
 add_executable(bar_test tests/bar_test.cpp)
 target_include_directories(bar_test PRIVATE src)
 target_link_libraries(bar_test PRIVATE PkgConfig::X11)
 add_executable(swallow_test tests/swallow_test.cpp)
 target_link_libraries(swallow_test PRIVATE PkgConfig::X11)
 add_executable(tray_test tests/tray_test.cpp)
 target_link_libraries(tray_test PRIVATE PkgConfig::X11)
 add_executable(stress_test tests/stress_test.cpp)
 target_link_libraries(stress_test PRIVATE PkgConfig::X11)
 option(PPWM_NESTED_TESTS "Run Xephyr integration tests (needs an X11 desktop)" OFF)
 if(PPWM_NESTED_TESTS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME nested_x11 COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/nested-test.py ${CMAKE_BINARY_DIR})
  set_tests_properties(nested_x11 PROPERTIES TIMEOUT 60)
 endif()
endif()
install(TARGETS ppwm RUNTIME DESTINATION bin)
